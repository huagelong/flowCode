### 9.4 Skill 自改进

```go
// skill_engine/improver.go

type SkillImprover struct {
    chatModel model.ChatModel
    store     SkillStore
}

type ImproveRequest struct {
    SkillID    uint
    Execution  ExecutionRecord
    Error      *TaskError        // nil 表示成功但可优化
}

func (s *SkillImprover) Improve(ctx context.Context, req ImproveRequest) error {
    if !s.needsImprovement(req) {
        return nil
    }
    analysis := s.analyze(ctx, req)
    improved := s.generateImproved(ctx, req.SkillID, analysis)
    if err := s.validate(improved); err != nil {
        return err
    }
    return s.store.Save(ctx, improved)
}

func (s *SkillImprover) needsImprovement(req ImproveRequest) bool {
    switch {
    case req.Error != nil:
        return true                    // 执行报错 → 必须改进
    case req.Execution.IsOutdated():
        return true                    // 内容过时 → 改进
    case req.Execution.Score < 0.6:
        return true                    // 质量评分低 → 改进
    default:
        return false
    }
}
```

**Skill 版本回滚机制**：

每次自改进自动保留历史版本，支持一键回滚：

```go
// skill_engine/rollback.go

// SaveWithHistory 保存改进版本，同时记录历史到 skill_generation_logs
func (s *SkillImprover) SaveWithHistory(ctx context.Context, skill *Skill, improved string) error {
    // 1. 当前版本写入历史（用于回滚）
    s.logRepo.Create(ctx, &SkillGenerationLog{
        SkillID:      skill.ID,
        SkillName:    skill.Name,
        SkillVersion: skill.Version,
        Definition:   skill.Definition, // 旧版本完整内容
        Status:       "improved",
    })

    // 2. 更新 Skill（version 递增）
    skill.Version++
    skill.Definition = improved
    return s.store.Save(ctx, skill)
}

// Rollback 回滚到指定版本
func (s *SkillImprover) Rollback(ctx context.Context, skillID uint, targetVersion int) error {
    skill, _ := s.store.GetByID(ctx, skillID)
    history, _ := s.logRepo.FindBySkillAndVersion(ctx, skillID, targetVersion)
    skill.Definition = history.Definition
    skill.Version = targetVersion
    return s.store.Save(ctx, skill)
}
```

| 操作 | 说明 |
|------|------|
| 自改进保存 | `SaveWithHistory` → 当前版本写入 `skill_generation_logs` → 新版本覆盖 |
| 手动回滚 | `Rollback(skillID, targetVersion)` → 从 `skill_generation_logs` 取回旧版本 |
| 历史查看 | 按 `skill_id` 查询 `skill_generation_logs`，按 `created_at` 排序 |
| 清理策略 | 保留最近 20 个版本，更早版本由定时任务归档 |
